// 1. Read full medicine list from MAIN SHEET (Columns I, J, K)
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("MAIN SHEET");
  var lastRow = sheet.getLastRow();
  var items = [];

  if (lastRow >= 4) {
    var values = sheet.getRange(4, 9, lastRow - 3, 3).getValues();
    for (var i = 0; i < values.length; i++) {
      var category = values[i][0] ? values[i][0].toString().trim() : "OTHER";
      var name = values[i][1] ? values[i][1].toString().trim() : "";
      var expiryRaw = values[i][2];

      if (name !== "") {
        var expiryStr = "";
        if (expiryRaw instanceof Date) {
          expiryStr = Utilities.formatDate(expiryRaw, Session.getScriptTimeZone(), "MMM yyyy");
        } else if (expiryRaw) {
          expiryStr = expiryRaw.toString().trim();
        }
        items.push({
          id: "sheet-row-" + (i + 4),
          category: category || "OTHER",
          name: name,
          expiry: expiryStr,
          unit: "box"
        });
      }
    }
  }

  return ContentService
    .createTextOutput(JSON.stringify({ status: "ok", count: items.length, items: items }))
    .setMimeType(ContentService.MimeType.JSON);
}

// 2. Append ordered items into "ORDER LIST" tab
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "ORDER LIST";
    var sheet = ss.getSheetByName(sheetName);

    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(["Date Ordered", "Category", "Medicine / Product Name", "Order Qty", "Unit", "Current Expiry", "Notes"]);
      sheet.getRange("A1:G1").setFontWeight("bold").setBackground("#D44D78").setFontColor("#ffffff");
      sheet.setFrozenRows(1);
    }

    var timestamp = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd hh:mm a");
    var items = data.items || [];
    items.forEach(function(item) {
      sheet.appendRow([
        timestamp,
        item.category || "",
        item.name || "",
        item.qty || 0,
        item.unit || "box",
        item.expiry || "",
        item.notes || ""
      ]);
    });

    return ContentService
      .createTextOutput(JSON.stringify({ status: "ok", added: items.length }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ status: "error", message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
